Interface IArrayPaginatorParams<FilterData, RowData>

Interface representing the parameters for array pagination.

interface IArrayPaginatorParams<
    FilterData extends {} = IAnything,
    RowData extends IRowData = IAnything,
> {
    chipsHandler?: (
        rows: RowData[],
        chips: Partial<Record<keyof RowData, boolean>>,
    ) => RowData[];
    compareFn?: (a: RowData, b: RowData, field: keyof RowData) => number;
    fallback?: (e: Error) => void;
    filterHandler?: (rows: RowData[], filterData: FilterData) => RowData[];
    onData?: (rows: RowData[], state: IState<FilterData, RowData>) => void;
    onLoadEnd?: (isOk: boolean) => void;
    onLoadStart?: () => void;
    paginationHandler?: (
        rows: RowData[],
        pagination: ListHandlerPagination,
    ) => RowData[];
    removeEmptyFilters?: (data: FilterData) => Partial<FilterData>;
    responseMap?: (
        json: RowData[],
    ) => Record<string, any>[] | Promise<Record<string, any>[]>;
    searchEntries?: string[];
    searchFilterChars?: string[];
    searchHandler?: (rows: RowData[], search: string) => RowData[];
    sortHandler?: (
        rows: RowData[],
        sort: ListHandlerSortModel<RowData>,
    ) => RowData[];
    withChips?: boolean;
    withFilters?: boolean;
    withPagination?: boolean;
    withSearch?: boolean;
    withSort?: boolean;
    withTotal?: boolean;
}
